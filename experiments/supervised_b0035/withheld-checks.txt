def test_admitted_refresh_preserves_native_and_current_inventory(tmp_path):
 async def run():
  h=make_host(tmp_path);state=progress();h.projection=ProjectionPolicy(10000,8000)
  first=inventory(await h.prepare(state))
  async def apply(changes):write_source_changes(h.scope.root,changes)
  h.effect=apply
  r=ModelResponse('',InteractionSource('oracle'),tool_calls=(ModelToolCall('read_source',json.dumps(dict(path='package/source.py',start=0)),'read'),))
  assert h.classify(r)=='acquire';await h.acquire(r);await h.prepare(state)
  r=ModelResponse('',InteractionSource('oracle'),tool_calls=(ModelToolCall('replace_source',json.dumps(dict(path='package/source.py',old_text='value =',new_text='changed =')),'edit'),))
  assert h.classify(r)=='edit';proposal=await h.propose(r);assert proposal
  await h.apply(proposal)
  # Actual fixed check and normal correlated feedback, no synthetic receipt.
  passed=await h.check(proposal);assert passed
  await h.feedback(r,proposal,passed=passed)
  q=await h.prepare(state);current=inventory(q)
  assert current['snapshot_id']==str(h.scope.snapshot.id)!=first['snapshot_id']
  rows={x['address']:x for x in current['resources']}
  raw=(tmp_path/'package/source.py').read_bytes()
  assert rows['package/source.py']['byte_size']==len(raw)
  assert rows['package/source.py']['content_identity']==str(h.scope.snapshot.resource_at(h.objective.editable[0]).content_identity)
  assert [m.tool_call_id for m in q.prompt.messages if m.role=='tool']==['read','edit']
  assert request_bytes(q)<=h.objective.prompt_bytes
 asyncio.run(run())
