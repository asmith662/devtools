from pathlib import Path

from devtools.core.time import Duration,Stopwatch
from devtools.resources.commands import Command,CommandExecutor,CommandOutputPolicy
root=Path.cwd();area=root/'.local/learning-integration-final/package'
bootstrap='import sys,runpy;sys.argv=sys.argv[1:];runpy.run_path(sys.argv[0],run_name="__main__")'
cmd=Command(r'C:\Python314\python.exe').args('-S','-c',bootstrap,str(area/'package_guard.py'),str(root),str(area/'host314/Lib/site-packages'),str(area/'host314_custody.py'),str(area/'host314-custody-01'),str(getattr(sys,'_base_executable'))).cwd(resolve_path(area/'independent-cwd')).with_environment({'PYTHONPATH':'','PYTHONDONTWRITEBYTECODE':'1'}).with_timeout(Duration.seconds(60))
