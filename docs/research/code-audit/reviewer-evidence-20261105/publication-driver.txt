def copy(name,source):

def ledger(cycle,grant):
 folder=cycle.routine.folder/'local'/grant['decision']['sha256']
 entries=cycle.routine.local.records(grant['reference'])
 results={p.name:_load(p,'backlog-routine-local-result-v1') for p in folder.glob('*-result.json')}
 assert {e['sha256']+'-result.json' for e in entries}==set(results)
 assert all(v['entry']+'.json'==n.removesuffix('-result.json')+'.json' for n,v in results.items())
 return {'grant':grant,'entries':entries,'results':results}
def raw(path):
 data=path.read_bytes();text=data.decode('utf-8');assert text.encode()==data
 return {'sha256':hashlib.sha256(data).hexdigest(),'bytes':len(data),'utf8':text}
