def inspect(root, content, name, lookup):
    root.mkdir(parents=True, exist_ok=True)
    address = RepositoryResourceAddress("probe.py")
    write(BinaryFile(resolve_path(root / address.value), content.encode("utf-8")), overwrite=False)
    snapshot = observe_repository_resources(repository=Repository.new(), root=resolve_path(root), addresses=(address,), maximum_resource_bytes=8192)
    module = interpret_python_module_resources(snapshot, module_root=PythonModuleRoot("."), resource_addresses=(address,)).interpretations[0]
    result = lookup(snapshot, module=module, declared_name=name)
    assert result.module == module
    assert result.function_analysis.derivation.dependency.snapshot_id == snapshot.id
    assert result.class_analysis.derivation.dependency.snapshot_id == snapshot.id
    return result, snapshot, address

def verify(root, row, lookup):
    result, snapshot, address = inspect(root, row["source"], row.get("name_to_lookup", "f"), lookup)
    actual = result.outcome.value
    if actual != row["expected"]:
        return {"case": row["name"], "expected": row["expected"], "actual": actual}
    if actual == "resolved":
        assert result.target is not None
        assert result.target.support.snapshot_id == snapshot.id
        assert result.target.support.resource_address == address
        assert result.target.support.source_range.start_line == row.get("line", 1)
        assert result.target.support.source_range.start_column_utf8 == 0
    else:
        assert result.target is None
    return None

@pytest.mark.parametrize("group", ["local-scopes", "wrapped-locals", "targets", "conservative"])
def test_declared_namespace_matrix(tmp_path, group):
    failures = []
    for row in CASES:
        if row["group"] == group:
            problem = verify(tmp_path / row["name"], row, lookup_python_module_declaration)
            if problem is not None:
                failures.append(problem)
    assert not failures, failures
