    'observed_failure': "PermissionError: [WinError 5] Access is denied: C:\\Users\\recoveryadmin\\AppData\\Local\\Temp\\pytest-of-recoveryadmin",

    'diagnostic': {'argv': ['.venv/Scripts/python.exe','-m','pytest','tests/agents/coding/compaction/test_adaptive.py','-x','--no-cov','--tb=short'],

for name in ('finish.py','verify_docs.py'):
