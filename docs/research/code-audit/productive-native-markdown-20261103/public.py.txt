"""Frozen public behavioral acceptance for literal ATX title hashes."""
from devtools.core.paths import ResolvedPath
from devtools.resources.filesystem import MarkdownFile

def test_literal_hash_is_part_of_heading_title(tmp_path):
    source = '# C#\nGuide\n## Topic##\nDetails\n'
    document = MarkdownFile(ResolvedPath(tmp_path / 'guide.md'), source)
    assert [(h.level, h.text, h.line) for h in document.headings] == [(1, 'C#', 1), (2, 'Topic##', 3)]
    assert document.find_heading('C#').line == 1
    assert document.find_section('Topic##').content == 'Details'
    assert document.content == source

def test_separated_closing_markers_still_normalize(tmp_path):
    document = MarkdownFile(ResolvedPath(tmp_path / 'guide.md'), '# C# ### \t\n## Title\t##\n# \n')
    assert tuple(h.text for h in document.headings) == ('C#', 'Title', '')
